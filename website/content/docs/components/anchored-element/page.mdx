import { AnchoredElement } from '@jarl/anchored-element';
import { Button } from '@jarl/button';

# AnchoredElement

## Basic usage

<PreviewBox>
  <AnchoredElement content="Anchored content">
    <span>Hover me</span>
  </AnchoredElement>
</PreviewBox>

```tsx
<AnchoredElement content="Anchored content">
  <span>Hover me</span>
</AnchoredElement>
```

## With arrow

<PreviewBox>
  <AnchoredElement withArrow content="Anchored content">
    <span>Hover me</span>
  </AnchoredElement>
</PreviewBox>

```tsx
<AnchoredElement withArrow content="Anchored content">
  <span>Hover me</span>
</AnchoredElement>
```

## Disable background scroll

<PreviewBox>
  <AnchoredElement content="Anchored content" disableBackgroundScroll>
    <span>Hover me</span>
  </AnchoredElement>
</PreviewBox>

```tsx
<AnchoredElement content="Anchored content" disableBackgroundScroll>
  <span>Hover me</span>
</AnchoredElement>
```

## Trigger

<PreviewBox>
  <AnchoredElement
    content="Anchored content"
    position="top"
    triggerEvents="click"
    withArrow
  >
    <Button>Click</Button>
  </AnchoredElement>
</PreviewBox>

```tsx
<AnchoredElement
  content="Anchored content"
  position="top"
  triggerEvents="click"
  withArrow
>
  <Button>Click</Button>
</AnchoredElement>
```

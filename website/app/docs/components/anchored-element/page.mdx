import { AnchoredElement } from '@jarl/anchored-element';

import '@jarl/styled-system/styles.css';

# AnchoredElement

## Basic usage

```tsx
<AnchoredElement content="Anchored content">
  <span>Hover me</span>
</AnchoredElement>
```

## With arrow

```tsx
<AnchoredElement withArrow content="Anchored content">
  <span>Hover me</span>
</AnchoredElement>
```

## Disable background scroll

```tsx
<AnchoredElement content="Anchored content" disableBackgroundScroll>
  <span>Hover me</span>
</AnchoredElement>
```

## Delay

```tsx
<AnchoredElement content="Anchored content" delay={1000}>
  <span>Hover me</span>
</AnchoredElement>
```

## Show/hide delays

```tsx
<AnchoredElement content="Anchored content" delay={[750, 200]}>
  <span>Hover me</span>
</AnchoredElement>
```

      <Button onClick={() => setModalInfo({ isOpened: true, justify: 'center', align: 'end' })}>Bottom center</Button>
      <Button onClick={() => setModalInfo({ isOpened: true, justify: 'start', align: 'start' })}>Top left</Button>
      <Button onClick={() => setModalInfo({ isOpened: true, justify: 'end', align: 'start' })}>Top right</Button>
      <Button onClick={() => setModalInfo({ isOpened: true, justify: 'start', align: 'center' })}>Center left</Button>
      <Button onClick={() => setModalInfo({ isOpened: true, justify: 'end', align: 'center' })}>Center right</Button>
      <Button onClick={() => setModalInfo({ isOpened: true, justify: 'start', align: 'end' })}>Bottom left</Button>
      <Button onClick={() => setModalInfo({ isOpened: true, justify: 'end', align: 'end' })}>Bottom right</Button>

      <Modal
        align={modalInfo.align}
        justify={modalInfo.justify}
        onCloseRequested={() => setModalInfo(prev => ({ ...prev, isOpened: false }))}

        <Modal.Backdrop />

          <Modal.CloseButton />
          <Modal.Header>
            <Modal.Title>Modal Title</Modal.Title>
          </Modal.Header>
          <Modal.Body>
